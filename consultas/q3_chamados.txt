# Consulta 3: chamados abertos em 2026, sem outros critérios. Com lista_ord, ordena pela data/hora de abertura, do mais antigo ao mais recente
# Com lista_gen, agrupa por unidade > grupo de atendimento > dia de abertura. 
# Com lista_cruz, cruza unidade (linha) x dia de abertura (coluna).
 
FILTER|ano_abertura|==|2026
 
# Ordenação cronológica (campo de data)
ORDER|data_hora_abertura|ASC
 
# Agrupamento em 3 níveis para a lista generalizada
GROUP|unidade|grupo|dia_abertura
 
# Cruzamento para a lista cruzada
CROSS|unidade|dia_abertura

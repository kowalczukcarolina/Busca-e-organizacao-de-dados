# Consulta 1: asteroides que passam muito perto (distância <= 0.05 UA) e com risco >= 0.2.
# Pontua a semelhança com um alvo de velocidade (5 km/s) e magnitude (22), descarta quem tiver similaridade < 0.40 e ordena do mais ao menos similar.


FILTER|distance_au|<=|0.05
FILTER|risk_score|>=|0.2

# Critérios de Similaridade (SIM: campo|alvo|tolerancia|peso)
SIM|velocity_km_s|5.0|3.0|1.5
SIM|absolute_magnitude|22.0|5.0|1.0

# Limiar mínimo de aceitação da similaridade
ACCEPT|0.40

# Ordenação para a lista ordenada ou resultados
ORDER|similarity|DESC

# Agrupamento para lista generalizada
GROUP|threat_category|year

# Cruzamento para lista cruzada
CROSS|threat_category|year
